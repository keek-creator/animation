local function stopCurrentEmote()
    if currentTrack and currentTrack.IsPlaying then
        currentTrack:Stop()
    end
    currentTrack = nil
end

local function resolveAnimId(id)
    -- Try to get the actual animation id from emote asset
    local ok, objects = pcall(function()
        return game:GetObjects("rbxassetid://" .. tostring(id))
    end)
    if ok and objects then
        local function findAnim(obj)
            if obj:IsA("Animation") then
                local raw = obj.AnimationId:gsub("rbxassetid://",""):gsub("http://www.roblox.com/asset/%?id=","")
                local n = tonumber(raw)
                if n and n > 0 then return n end
            end
            for _, c in ipairs(obj:GetChildren()) do
                local f = findAnim(c)
                if f then return f end
            end
        end
        for _, obj in ipairs(objects) do
            pcall(function() obj.Parent = workspace end)
            local found = findAnim(obj)
            task.delay(1, function() pcall(function() obj:Destroy() end) end)
            if found then return found end
        end
    end
    return tonumber(id)
end

local function playEmote(emote)
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChild("Humanoid")
    if not hum then return end
    local animator = hum:FindFirstChild("Animator")
    if not animator then return end

    stopCurrentEmote()

    task.spawn(function()
        local animId = resolveAnimId(emote.id)
        if not animId then return end

        local anim = Instance.new("Animation")
        anim.AnimationId = "rbxassetid://" .. tostring(animId)

        local ok, track = pcall(function()
            return animator:LoadAnimation(anim)
        end)
        if not ok or not track then return end

        track:Play()
        track:AdjustSpeed(emoteSpeed)
        currentTrack = track

        if freezeEnabled then
            local origWalk = hum.WalkSpeed
            local origJump = hum.JumpPower
            hum.WalkSpeed = 0
            hum.JumpPower = 0
            track.Stopped:Connect(function()
                if freezeEnabled then
                    hum.WalkSpeed = origWalk
                    hum.JumpPower = origJump
                end
            end)
        end
    end)
end

-- RENDER EMOTE LIST
-- ─────────────────────────────────────────
local function clearList()
    for _, child in ipairs(scrollFrame:GetChildren()) do
        if child:IsA("Frame") or child:IsA("TextButton") then
            child:Destroy()
        end
    end
end

local function renderPage()
    clearList()
    local totalPages = math.max(1, math.ceil(#filteredEmotes / EMOTES_PER_PAGE))
    if currentPage > totalPages then currentPage = totalPages end
    if currentPage < 1 then currentPage = 1 end

    pageLabel.Text = "Page " .. currentPage .. " / " .. totalPages
    prevBtn.TextTransparency = currentPage == 1 and 0.5 or 0
    nextBtn.TextTransparency = currentPage == totalPages and 0.5 or 0

    local startIdx = (currentPage - 1) * EMOTES_PER_PAGE + 1
    local endIdx = math.min(startIdx + EMOTES_PER_PAGE - 1, #filteredEmotes)

    for i = startIdx, endIdx do
        local emote = filteredEmotes[i]
        if not emote then break end

        local row = Instance.new("TextButton")
        row.Size = UDim2.new(1, 0, 0, 38)
        row.BackgroundColor3 = (selectedEmote and selectedEmote.id == emote.id)
            and Color3.fromRGB(10, 40, 60)
            or Color3.fromRGB(11, 19, 30)
        row.TextColor3 = Color3.fromRGB(190, 215, 235)
        row.Font = Enum.Font.Gotham
        row.TextSize = 13
        row.Text = ""
        row.LayoutOrder = i
        row.BorderSizePixel = 0
        row.ZIndex = 7
        row.Parent = scrollFrame
        makeCorner(row, 5)

        if selectedEmote and selectedEmote.id == emote.id then
            makeStroke(row, Color3.fromRGB(0, 180, 220), 1)
        end

        -- Emote name label
        local nameLabel = Instance.new("TextLabel")
        nameLabel.Size = UDim2.new(1, -110, 1, 0)
        nameLabel.Position = UDim2.new(0, 10, 0, 0)
        nameLabel.BackgroundTransparency = 1
        nameLabel.TextColor3 = Color3.fromRGB(190, 215, 235)
        nameLabel.Font = Enum.Font.Gotham
        nameLabel.TextSize = 13
        nameLabel.TextXAlignment = Enum.TextXAlignment.Left
        nameLabel.Text = emote.name
        nameLabel.ZIndex = 8
        nameLabel.Parent = row

        -- ID label
        local idLabel = Instance.new("TextLabel")
        idLabel.Size = UDim2.new(0, 100, 1, 0)
        idLabel.Position = UDim2.new(1, -108, 0, 0)
        idLabel.BackgroundTransparency = 1
        idLabel.TextColor3 = Color3.fromRGB(50, 90, 120)
        idLabel.Font = Enum.Font.Gotham
        idLabel.TextSize = 10
        idLabel.TextXAlignment = Enum.TextXAlignment.Right
        idLabel.Text = tostring(emote.id)
        idLabel.ZIndex = 8
        idLabel.Parent = row

        -- Play button
        local playBtn = Instance.new("TextButton")
        playBtn.Size = UDim2.new(0, 52, 0, 26)
        playBtn.Position = UDim2.new(1, -58, 0.5, -13)
        playBtn.BackgroundColor3 = Color3.fromRGB(0, 60, 90)
        playBtn.TextColor3 = Color3.fromRGB(0, 220, 255)
        playBtn.Font = Enum.Font.GothamBold
        playBtn.TextSize = 11
        playBtn.Text = "▶ PLAY"
        playBtn.ZIndex = 9
        playBtn.Parent = row
        makeCorner(playBtn, 4)

        -- Hover
        row.MouseEnter:Connect(function()
            if not (selectedEmote and selectedEmote.id == emote.id) then
                TweenService:Create(row, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(14, 28, 44)}):Play()
            end
        end)
        row.MouseLeave:Connect(function()
            if not (selectedEmote and selectedEmote.id == emote.id) then
                TweenService:Create(row, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(11, 19, 30)}):Play()
            end
        end)

        -- Click row = select
        row.MouseButton1Click:Connect(function()
            selectedEmote = emote
            renderPage()
        end)

        -- Click play = play immediately
        playBtn.MouseButton1Click:Connect(function()
            selectedEmote = emote
            playEmote(emote)
            renderPage()
        end)
    end

    scrollFrame.CanvasSize = UDim2.new(0, 0, 0, listLayout.AbsoluteContentSize.Y + 8)
    scrollFrame.CanvasPosition = Vector2.new(0, 0)
end

-- ─────────────────────────────────────────
-- SEARCH
-- ─────────────────────────────────────────
local searchDebounce = nil
searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    if searchDebounce then task.cancel(searchDebounce) end
    searchDebounce = task.delay(0.25, function()
        local query = searchBox.Text:lower():gsub("^%s+",""):gsub("%s+$","")
        if query == "" then
            filteredEmotes = allEmotes
        else
            filteredEmotes = {}
            for _, e in ipairs(allEmotes) do
                if e.name:lower():find(query, 1, true) then
                    table.insert(filteredEmotes, e)
                end
            end
        end
        countLabel.Text = #filteredEmotes .. " results"
        currentPage = 1
        renderPage()
    end)
end)

-- ─────────────────────────────────────────
-- DRAGGING (title bar)
-- ─────────────────────────────────────────
do
    local dragging = false
    local dragStart, startPos

    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = mainFrame.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            mainFrame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

-- ─────────────────────────────────────────
-- LOAD EMOTES FROM GITHUB
-- ─────────────────────────────────────────
task.spawn(function()
    local ok, result = pcall(function()
        return game:HttpGet("https://raw.githubusercontent.com/7yd7/sniper-Emote/refs/heads/test/EmoteSniper.json")
    end)
    if ok and result then
        local dok, data = pcall(function()
            return HttpService:JSONDecode(result)
        end)
        if dok and data and data.data then
            allEmotes = data.data
            filteredEmotes = allEmotes
            countLabel.Text = #allEmotes .. " emotes"
            renderPage()
        else
            countLabel.Text = "failed to parse"
        end
    else
        countLabel.Text = "failed to load"
    end
end)

    humanoid = char:WaitForChild("Humanoid")
    currentTrack = nil
end)
