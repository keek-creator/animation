-- ══════════════════════════════════════════════════════════════════════
--  PLAYBACK
-- ══════════════════════════════════════════════════════════════════════
local function unfreezeChar()
    local char = player.Character; if not char then return end
    local hum = char:FindFirstChild("Humanoid"); if not hum then return end
    hum.WalkSpeed = 16; hum.JumpPower = 50
end

local function stopEmote()
    if currentTrack then
        pcall(function() currentTrack:Stop() end)
        currentTrack = nil
    end
    if not freezeEnabled then unfreezeChar() end
end

local function resolveAnimId(id)
    local ok, objects = pcall(function()
        return game:GetObjects("rbxassetid://" .. tostring(id))
    end)
    if ok and objects then
        local function findAnim(obj)
            if obj:IsA("Animation") then
                local raw = obj.AnimationId
                    :gsub("rbxassetid://", "")
                    :gsub("http://www.roblox.com/asset/%?id=", "")
                local n = tonumber(raw)
                if n and n > 0 then return n end
            end
            for _, c in ipairs(obj:GetChildren()) do
                local f = findAnim(c); if f then return f end
            end
        end
        for _, obj in ipairs(objects) do
            pcall(function() obj.Parent = workspace end)
            local found = findAnim(obj)
            task.delay(1, function() pcall(function() obj:Destroy() end) end)
            if found then return found end
        end
    end
    return tonumber(id)
end

local function playItem(item)
    local char = player.Character; if not char then return end
    local hum  = char:FindFirstChild("Humanoid"); if not hum then return end
    local anim = hum:FindFirstChild("Animator"); if not anim then return end
    stopEmote()
    task.spawn(function()
        local animId = resolveAnimId(item.id)
        if not animId then return end
        local a = Instance.new("Animation")
        a.AnimationId = "rbxassetid://" .. tostring(animId)
        local ok2, track = pcall(function() return anim:LoadAnimation(a) end)
        if not ok2 or not track then return end
        track:Play()
        track:AdjustSpeed(emoteSpeed)
        currentTrack = track
        if freezeEnabled then
            hum.WalkSpeed = 0; hum.JumpPower = 0
            track.Stopped:Connect(function()
                if freezeEnabled then unfreezeChar() end
            end)
        end
    end)
end

-- stop emote when player moves
local function setupMoveStop(char)
    if moveConn then moveConn:Disconnect(); moveConn = nil end
    local hum = char:FindFirstChildOfClass("Humanoid"); if not hum then return end
    moveConn = hum:GetPropertyChangedSignal("MoveDirection"):Connect(function()
        if hum.MoveDirection.Magnitude > 0 and currentTrack then
            stopEmote()
            unfreezeChar()
        end
    end)
end
setupMoveStop(character)

    nextBtn.TextTransparency = currentPage == total and 0.55 or 0

local function refreshTabs()
    if currentTab == "emotes" then
        tabEmoteBtn.BackgroundColor3 = Color3.fromRGB(0, 52, 82)
        tabEmoteBtn.TextColor3       = Color3.fromRGB(0, 205, 255)
        tabAnimBtn.BackgroundColor3  = Color3.fromRGB(11, 22, 35)
        tabAnimBtn.TextColor3        = Color3.fromRGB(60, 100, 130)
    else
        tabAnimBtn.BackgroundColor3  = Color3.fromRGB(0, 52, 82)
        tabAnimBtn.TextColor3        = Color3.fromRGB(0, 205, 255)
        tabEmoteBtn.BackgroundColor3 = Color3.fromRGB(11, 22, 35)
        tabEmoteBtn.TextColor3       = Color3.fromRGB(60, 100, 130)
    end
end

local function switchTab(tab)
    currentTab  = tab
    currentPage = 1
    searchBox.Text = ""
    filtEmotes = allEmotes
    filtAnims  = allAnimations
    local list = getList()
    statusLbl.Text = #list > 0 and (#list .. " loaded") or "loading..."
    refreshTabs()
    renderPage()
end

tabEmoteBtn.MouseButton1Click:Connect(function() switchTab("emotes") end)
tabAnimBtn.MouseButton1Click:Connect(function()  switchTab("animations") end)
refreshTabs()

-- ══════════════════════════════════════════════════════════════════════
local searchDebounce
searchBox:GetPropertyChangedSignal("Text"):Connect(function()
    if searchDebounce then task.cancel(searchDebounce) end
    searchDebounce = task.delay(0.2, function()
        local q = searchBox.Text:lower():gsub("^%s+",""):gsub("%s+$","")
        local src = currentTab == "emotes" and allEmotes or allAnimations
        local out = {}
        if q == "" then
            out = src
        else
            for _, v in ipairs(src) do
                if v.name:lower():find(q, 1, true) then out[#out+1] = v end
            end
        end
        if currentTab == "emotes" then
            filtEmotes = out
            statusLbl.Text = #out .. " emotes"
        else
            filtAnims = out
            statusLbl.Text = #out .. " animations"
        end
        currentPage = 1
        renderPage()
    end)
end)

        freezeBtn.Text             = "❄ FREEZE: ON"
        freezeBtn.BackgroundColor3 = Color3.fromRGB(0, 38, 25)
        freezeBtn.TextColor3       = Color3.fromRGB(0, 255, 145)
    else
        freezeBtn.Text             = "❄ FREEZE: OFF"
        freezeBtn.BackgroundColor3 = Color3.fromRGB(7, 22, 15)
        freezeBtn.TextColor3       = Color3.fromRGB(0, 195, 115)
        unfreezeChar()
    end
end)

stopBtn.MouseButton1Click:Connect(function()
    stopEmote()
    unfreezeChar()
end)

-- ══════════════════════════════════════════════════════════════════════
--  OPEN / CLOSE
-- ══════════════════════════════════════════════════════════════════════
local function setOpen(open)
    guiOpen = open
    if open then
        panel.Visible = true
        panel.BackgroundTransparency = 0
        pillIcon.Text      = "✕"
        pillIcon.TextColor3 = Color3.fromRGB(255, 75, 75)
        tw(pillBg, 0.18, {BackgroundColor3 = Color3.fromRGB(25, 6, 10)})
        pillStroke.Color   = Color3.fromRGB(255, 80, 80)
    else
        tw(panel, 0.14, {BackgroundTransparency = 1})
        task.delay(0.15, function()
            if not guiOpen then panel.Visible = false end
        end)
        pillIcon.Text       = "▶"
        pillIcon.TextColor3 = Color3.fromRGB(0, 215, 255)
        tw(pillBg, 0.18, {BackgroundColor3 = Color3.fromRGB(5, 12, 22)})
        pillStroke.Color    = Color3.fromRGB(0, 200, 255)
    end
end

pillBtn.MouseButton1Click:Connect(function() setOpen(not guiOpen) end)
closeBtn.MouseButton1Click:Connect(function() setOpen(false) end)

-- ══════════════════════════════════════════════════════════════════════
--  DRAG (title bar)
-- ══════════════════════════════════════════════════════════════════════
do
    local dragging, dragStart, startPos = false, nil, nil
    titleBar.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true; dragStart = inp.Position; startPos = panel.Position
        end
    end)
    table.insert(connections, UserInputService.InputChanged:Connect(function(inp)
        if dragging and inp.UserInputType == Enum.UserInputType.MouseMovement then
            local d = inp.Position - dragStart
            panel.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end))
    table.insert(connections, UserInputService.InputEnded:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
    end))
end

task.spawn(function()
    local ok1, r1 = pcall(function()
        return game:HttpGet("https://raw.githubusercontent.com/7yd7/sniper-Emote/refs/heads/test/EmoteSniper.json")
    end)
    if ok1 and r1 then
        local ok2, data = pcall(function() return HttpService:JSONDecode(r1) end)
        if ok2 and data and data.data then
            allEmotes  = data.data
            filtEmotes = allEmotes
            if currentTab == "emotes" then
                statusLbl.Text = #allEmotes .. " emotes"
                renderPage()
            end
        end
    end

    local ok3, r3 = pcall(function()
        return game:HttpGet("https://raw.githubusercontent.com/7yd7/sniper-Emote/refs/heads/test/AnimationSniper.json")
    end)
    if ok3 and r3 then
        local ok4, data = pcall(function() return HttpService:JSONDecode(r3) end)
        if ok4 and data and data.data then
            allAnimations = data.data
            filtAnims     = allAnimations
        end
    end

    if currentTab == "emotes" and #allEmotes > 0 then
        statusLbl.Text = #allEmotes .. " emotes"
    end
    renderPage()
end)
